// Compiled only into the CI copy of the pinned historical helper. No IPC or
// network endpoint invokes the fixture, and the packaged service excludes it.
namespace PackProof.Connect;

public sealed partial class AgentRuntime
{
    internal void SeedHistoricalFixture()
    {
        RequireLocalInitialization();
        if (host.Environment != "development" || state.Token != null || queue.Entries().Count != 0)
            throw new InvalidOperationException("Historical fixture requires a fresh unenrolled development store.");
        var bytes = "Reference,Description,Quantity\r\n0000123,CI synthetic pending item,2\r\n"u8.ToArray();
        var config = Json.Element(new { integrationAccountId = "ci-fixture", operatorMembershipId = "ci-fixture",
            sourceName = "Synthetic pending report", timeZone = "UTC", mapping = new { shipmentReference = "Reference", itemTitle = "Description", quantity = "Quantity" } });
        var entry = queue.Stage(bytes, "pending.csv", new(config), "ci-historical-credential", "ci-historical-config", false);
        var payload = queue.Payload(entry);
        payload.Receipt = Json.Element(new { id = "ci-pending-receipt", state = "previewed", original = new { sha256 = entry.Digest } });
        queue.Save(entry, payload, "awaiting_review", "ci-pending-receipt");
        queue.Checkpoint();
        Save();
    }
}
